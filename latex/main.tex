 \begin{frame}{Key Questions}
  
  \begin{enumerate}
    \setlength\itemsep{0.8em}
    \item Conglomerate mergers are not well understood in economics. Why would firms in unrelated markets merge?
    \item How can antitrust authorities, who are tasked with protecting consumers from harmful mergers, evaluate these mergers?
    \item How will transaction costs (communication, transfer, trust, etc.) affect conglomerate structure?
  \end{enumerate}

\end{frame}


\begin{frame}{Model and setup}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{minipage}{\linewidth}
        \begin{algorithm}[H]
          \SetAlgoLined
          \KwIn{$N, M, C, \Delta, T, p , \alpha, \lambda$}
          \For{$t$ in $T$}{
            \ForEach{$m_i$ $\in$ $N \times M$}{
              Merge with $m_j$ if $m_j$ $\notin$ $n_i$ and $m_j$ $\notin$ $c_{it}$ with $p$ 
            }
            \ForEach{$c_k \in C$}{
              $pool_k$ = $\sum_{m_i \in c_k} \alpha \Delta_{it} - \lambda(c_k)$
              $\Delta_{it}^{*} = (1 - \alpha) \Delta_{it} + \frac{pool_k}{\# m_i \in c_k}$
            }
            \ForEach{$m_i$ $\in$ $N \times M$}{
              Split if $s_{it} < s_{it - 4}$  
            }

          }
        \end{algorithm}
      \end{minipage}
    \end{column}
    \begin{column}{0.5\textwidth}  
      \begin{itemize}
        \item $N = \#$ Markets (100)
        \item $M = \#$ Firms per Market (100)
        \item $C =$ Conglomerates
        \item $\Delta =$ Growth $(\sim \mathcal{N}(\mu_n, \sigma_n))$
        \item $T =$ Total steps (1000)
        \item $p =$ Merge probability (0.02)
        \item $\alpha =$ Pooling rate ($0.2$)
        \item $s =$ Market share
        \item $\lambda =$ Transaction cost function
        \begin{itemize}
          \item $\frac{\partial \lambda}{\partial \# m_i \in c_k} > 0$
          \item $\frac{\partial^2 \lambda }{\partial^2 \# m_i \in c_k} > 0$ 
        \end{itemize}
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

  \begin{itemize}
    \setlength\itemsep{0.8em}
    \item Merging is a dominant strategy with more than 70\,\% of firms in a conglomerate (invariant to merge probability $p$)
    \item Stabilization in the number of conglomerates (convergence speed depends on transaction costs)
    \item Highly unequal distribution of conglomerate size, with the 90\,\% and 70\,\% quantiles being far apart from the 25\,\% and 10\,\% ones
  \end{itemize}

\begin{frame}{Market Concentration (1/3) - Size Matters}

    \begin{column}{0.3\textwidth}

    \end{column}

    \begin{column}{0.3\textwidth}
      \centering
      \includegraphics[width=\textwidth]{figures/est.png}
    \end{column}
  \end{columns}

  \begin{itemize}
    \setlength\itemsep{0.8em}
    \item Increasing the conglomerate size decreases the chance to end up in the lower quantiles (ranks) and to have lower market share ($\frac{s_{it}}{\sum_{i} s_{it}}$)
    \item Increasing the size above a certain threshold will not increase the success anymore (inverse U-shape)
    \item In the progression of the model the optimal size increases, but the costs eat up the gains from size at an increasing rate
  \end{itemize}

\end{frame}

\begin{frame}{Key Findings}
  
  \begin{enumerate}
    \setlength\itemsep{0.8em}
    \item Including dynamic frameworks allows for the explanation of many phenomena of economic exchange such as conglomerate formation
    \item Increasing firm cooperation through conglomerate formation does not necessarily harm competition
    \item Transaction costs play a crucial role in determining the size of a conglomerate
  \end{enumerate}

\end{frame}

            }
            \ForEach{$m_i$ $\in$ $N \times M$}{
              Split if $s_{it} < s_{it - 4}$  
            }
            }
          }
            \only<2>{
              \vdots
              \ForEach{$m \in M$}{
                $p_{mt} = \pi(\cdot)$
              }
              \ForEach{$c_k \in C$}{
                $pool_k$ = $\sum_{m_i \in c_k} \alpha \textcolor{red}{\Pi_{it}(\cdot)}  - \lambda(c_k)$
                $\Delta_{it}^{*} = (1 - \alpha) \Delta_{it} + \frac{pool_k}{\# m_i \in c_k}$
              }
              \ForEach{$m \in M$}{
                $s_{mt}(\boldsymbol{\Pi})$
              }

            \vdots
            }
          
        \end{algorithm}
      \end{minipage}
    \end{column}
    \begin{column}{0.5\textwidth}  
      \only<2>{\begin{itemize}
        \item Allow firms to set prices following their policy $\pi(\cdot)$
        \item Fluctuations in a company's growth through stochastic costs $c_{it}$
        \item Market share as the cumulative profits $\Pi_{it} (\mathbf{P_t}, s_{it}) = D(\mathbf{P_t}, s_{it}) * p_{it} - c_{it}$ with $s_{it}(\boldsymbol{\Pi}) = \frac{\sum_{\tau=1}^{t}\Pi_{i \tau}}{\sum_{\tau=1}^{t} \underset{j: j \neq i}{\sum} \Pi_{j \tau}}$

      \end{itemize}
      }

\begin{frame}{Model Extension: Pricing Decision}

  \begin{columns}[T] % T aligns the tops of the columns
    \begin{column}{0.55\textwidth}
      \begin{itemize}
        \setlength\itemsep{0.8em}
        \item Demand function of the form $D(\mathbf{P_t}, s_{t}) = 1 - (1 - s_{it}) p_{it} + 0.5 \underset{j: j \neq i}{\max} \, p_{jt}$
        \item Learn the optimal pricing policy $\pi(\cdot)$ using a Proximal Policy Optimization reinforcement learning algorithm
        \item Information set of the firm  $\{p_{it-1}, s_{it-1}, \underset{j: j \neq i}{\max} \, p_{jt-1}, \underset{j: j \neq i}{\max} \, p_{jt-2}\}$
        \item Goal of the firm is to optimize the profit $\Pi_{it} (\mathbf{P_t}, s_{it}) = D(\mathbf{P_t}, s_{it}) p_{it}$
                
      \end{itemize}
    \end{column}
